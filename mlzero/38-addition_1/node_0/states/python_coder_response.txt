```python
"""
Skin Lesion Malignancy Prediction Script using AutoGluon MultiModal

This script trains an image classification model to predict the probability of malignancy for skin lesion images.
It uses AutoGluon's MultiModalPredictor for binary classification (malignant vs. benign).
- Loads and preprocesses the data (removes NA labels, drops index column).
- Trains the model, holding out 10% for validation if no explicit validation set is provided.
- Saves the trained model to a timestamped folder in the specified output directory.
- Predicts malignancy probability for each test image, preserving original indices and output format.
- Provides a function to predict on a new folder of images.
- Performs validation checks on the output.
- Prints AUROC on the held-out validation set.

# Installation (if needed):
# !pip install autogluon.multimodal --quiet

Usage:
- Place this script in your working environment.
- Ensure data is available at /home/anri21/be-fair/mlzero/addition_1_data
- Run the script. Results will be saved in /home/anri21/be-fair/mlzero/38-addition_1/node_0/output

Author: AutoML Agent
"""

import os
import shutil
import random
import time
import pandas as pd
import numpy as np

from autogluon.multimodal import MultiModalPredictor
from sklearn.metrics import roc_auc_score

# Set constants for paths
DATA_DIR = "/home/anri21/be-fair/mlzero/addition_1_data"
OUTPUT_DIR = "/home/anri21/be-fair/mlzero/38-addition_1/node_0/output"
TRAIN_CSV = os.path.join(DATA_DIR, "mydataset.csv")
IMAGE_DIR = DATA_DIR  # images are referenced by image_name column, assumed to be in DATA_DIR

# Ensure output directory exists
os.makedirs(OUTPUT_DIR, exist_ok=True)

def get_timestamp_folder(base_dir):
    """Generate a random timestamped folder path under base_dir."""
    timestamp = int(time.time()) + random.randint(0, 99999)
    folder = os.path.join(base_dir, f"model_{timestamp}")
    return folder

def get_image_path(row):
    """Return absolute path to image given a row with image_name."""
    return os.path.join(IMAGE_DIR, row['image_name'])

def map_label_to_binary(label):
    """Map label to binary: malignant=1, else=0."""
    return 1 if str(label).strip().lower() == "malignant" else 0

def main():
    # 1. Data Loading and Preprocessing
    df = pd.read_csv(TRAIN_CSV)
    # Remove unnecessary index column if present
    if 'Unnamed: 0' in df.columns:
        df = df.drop(columns=['Unnamed: 0'])
    # Remove training samples without valid labels (drop NA in 'label')
    df = df.dropna(subset=['label'])
    # Map label to binary (malignant=1, else=0)
    df['label'] = df['label'].apply(map_label_to_binary)
    # Add absolute image path column for AutoGluon
    df['image_path'] = df.apply(get_image_path, axis=1)
    # Check for missing images
    missing_images = df[~df['image_path'].apply(os.path.exists)]
    if not missing_images.empty:
        raise FileNotFoundError(f"Missing image files: {missing_images['image_path'].tolist()}")

    # 2. Train/Validation Split (10% holdout)
    df = df.sample(frac=1, random_state=42).reset_index(drop=True)  # shuffle
    val_frac = 0.1
    val_size = int(len(df) * val_frac)
    if val_size == 0 and len(df) > 1:
        val_size = 1  # ensure at least one sample for validation if possible
    train_df = df.iloc[:-val_size] if val_size > 0 else df
    val_df = df.iloc[-val_size:] if val_size > 0 else pd.DataFrame(columns=df.columns)

    # 3. Model Training
    model_dir = get_timestamp_folder(OUTPUT_DIR)
    os.makedirs(model_dir, exist_ok=True)
    predictor = MultiModalPredictor(
        label='label',
        problem_type='binary',
        eval_metric='roc_auc',
        path=model_dir
    )
    # Only use image_path and label for training
    train_data = train_df[['image_path', 'label']]
    predictor.fit(
        train_data=train_data,
        presets='best_quality',
        time_limit=None,
        seed=42
    )
    # Save the model (already saved by predictor.fit, but ensure it's in the right place)
    predictor.save(model_dir)

    # 4. Prediction on Test Set
    # For this task, we assume test images are provided as a folder path.
    # We'll implement a function to predict on a folder of new images.

    def predict_on_folder(image_folder):
        """
        Predict malignancy probability for each image in the given folder.
        Returns a DataFrame with columns: ['image_name', 'malignancy_probability']
        """
        # List all files in the folder (assume all are images)
        image_files = sorted([
            f for f in os.listdir(image_folder)
            if os.path.isfile(os.path.join(image_folder, f))
        ])
        test_df = pd.DataFrame({'image_name': image_files})
        test_df['image_path'] = test_df['image_name'].apply(lambda x: os.path.join(image_folder, x))
        # Predict probabilities
        preds = predictor.predict_proba(test_df[['image_path']])
        # For binary, predictor.predict_proba returns two columns: 0 and 1
        # We want the probability of class 1 (malignant)
        if 1 in preds.columns:
            malignancy_prob = preds[1].values
        else:
            # Sometimes columns are strings
            malignancy_prob = preds['1'].values
        result_df = pd.DataFrame({
            'image_name': test_df['image_name'],
            'malignancy_probability': malignancy_prob
        })
        return result_df

    # 5. Save predictions on test set (if test set is provided)
    # For this script, we assume test images are in a folder named "test_images" under DATA_DIR
    TEST_IMAGE_FOLDER = os.path.join(DATA_DIR, "test_images")
    if os.path.isdir(TEST_IMAGE_FOLDER):
        results_df = predict_on_folder(TEST_IMAGE_FOLDER)
        # Save results with the same format as test input (CSV)
        results_path = os.path.join(OUTPUT_DIR, "results.csv")
        results_df.to_csv(results_path, index=False)
        # Validation checks
        # 1. Indices preserved (image_name order matches test set)
        test_files = sorted([
            f for f in os.listdir(TEST_IMAGE_FOLDER)
            if os.path.isfile(os.path.join(TEST_IMAGE_FOLDER, f))
        ])
        assert list(results_df['image_name']) == test_files, "Image names/order mismatch!"
        # 2. Correct columns
        assert list(results_df.columns) == ['image_name', 'malignancy_probability'], "Column names mismatch!"
        # 3. Output format
        assert results_path.endswith('.csv'), "Output file format mismatch!"
        # 4. Number of predictions
        assert len(results_df) == len(test_files), "Number of predictions does not match number of test samples!"
        # 5. Sanity check: probabilities in [0,1]
        assert np.all((results_df['malignancy_probability'] >= 0) & (results_df['malignancy_probability'] <= 1)), "Probabilities out of range!"
        print(f"Predictions saved to {results_path}")

    # 6. Validation on held-out set
    if not val_df.empty:
        try:
            val_data = val_df[['image_path', 'label']]
            val_preds = predictor.predict_proba(val_data[['image_path']])
            # Get probability of class 1 (malignant)
            if 1 in val_preds.columns:
                val_probs = val_preds[1].values
            else:
                val_probs = val_preds['1'].values
            val_labels = val_data['label'].values
            val_auc = roc_auc_score(val_labels, val_probs)
            print(f"Validation AUROC: {val_auc:.4f}")
        except Exception as e:
            print(f"Validation failed: {e}")

    # 7. Provide the prediction function for new images
    # Save this function to the output directory for user import
    import dill
    func_path = os.path.join(OUTPUT_DIR, "predict_on_folder.dill")
    with open(func_path, "wb") as f:
        dill.dump(predict_on_folder, f)
    print(f"Prediction function saved to {func_path}")

if __name__ == "__main__":
    main()
```

- The prediction function `predict_on_folder` is saved as a dill-serialized object for later use.
