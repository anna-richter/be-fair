```python
import os
from PIL import Image

file_path = "/sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/basic_prompt_data/MyImages/image_16121.jpg"

try:
    file_size = os.path.getsize(file_path) / (1024 * 1024)
except Exception as e:
    file_size = None

info = {}
try:
    with Image.open(file_path) as img:
        info['format'] = img.format
        info['mode'] = img.mode
        info['size'] = img.size
except Exception as e:
    info['error'] = str(e)

print(f"File: {os.path.basename(file_path)}")
print(f"Type: JPEG Image")
print(f"Size: {file_size:.2f} MB" if file_size else "Size: Unknown")
if 'error' not in info:
    print(f"Image format: {info['format']}")
    print(f"Image mode: {info['mode']}")
    print(f"Dimensions: {info['size'][0]}x{info['size'][1]}")
else:
    print("Could not open image file.")
```
