```bash
#!/usr/bin/env bash
set -e

# Paths
ENV_DIR="/home/anri21/be-fair/mlzero/33-basic_prompt/node_6/conda_env"
PYTHON_VERSION="3.11"
REQ_COMMON="/home/anri21/.conda/envs/mlzero_env/lib/python3.12/site-packages/autogluon/assistant/tools_registry/_common/requirements.txt"
REQ_ML="/home/anri21/.conda/envs/mlzero_env/lib/python3.12/site-packages/autogluon/assistant/tools_registry/machine learning/requirements.txt"
PY_SCRIPT="/home/anri21/be-fair/mlzero/33-basic_prompt/node_6/generated_code.py"

# 1. Create conda env if not exists
if [ ! -d "$ENV_DIR" ]; then
    conda create -y -p "$ENV_DIR" python=$PYTHON_VERSION
fi

# 2. Activate env
eval "$(conda shell.bash hook)"
conda activate "$ENV_DIR"

# 3. Install uv if not present
if ! python -m pip show uv &>/dev/null; then
    python -m pip install uv
fi

# 4. Install requirements with uv (prerelease allowed)
uv pip install --prerelease=allow -r "$REQ_ML" -r "$REQ_COMMON"

# 5. Install extra packages needed for the script
python -m pip install pandas scikit-learn torch torchvision pillow joblib

# 6. Run the Python script
python "$PY_SCRIPT"
```

**Instructions:**
- Save as `run_node6.sh`, `chmod +x run_node6.sh`, then run with `./run_node6.sh`.
- If `conda` is not initialized in your shell, add `source ~/.bashrc` before the `eval` line.
