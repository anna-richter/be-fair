import os

file_path = "/sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/addition_3_data/descriptions.txt"

def show_text_file(fp, max_chars=768):
    with open(fp, 'r', encoding='utf-8', errors='replace') as f:
        content = f.read(max_chars)
    print(content)

def main():
    if not os.path.isfile(file_path):
        print("File not found.")
        return
    try:
        import pandas as pd
        try:
            df = pd.read_csv(file_path, sep=None, engine='python', nrows=3)
            cols = list(df.columns)
            ncol = len(cols)
            if ncol > 20:
                display_cols = cols[:10] + cols[-10:]
            else:
                display_cols = cols
            print("Columns:", display_cols)
            def trunc(x): return str(x)[:50] + ('...' if len(str(x)) > 50 else '')
            print(df.applymap(trunc).to_string(index=False))
            return
        except Exception:
            pass
    except ImportError:
        pass
    show_text_file(file_path)

main()
