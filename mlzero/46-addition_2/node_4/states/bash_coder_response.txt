```bash
#!/bin/bash
set -e

# Variables
ENV_DIR="/sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/46-addition_2/node_4/conda_env"
PYTHON_VERSION="3.11"
REQ_COMMON="/home/anri21/.conda/envs/mlzero_env/lib/python3.11/site-packages/autogluon/assistant/tools_registry/_common/requirements.txt"
REQ_ML="/home/anri21/.conda/envs/mlzero_env/lib/python3.11/site-packages/autogluon/assistant/tools_registry/machine learning/requirements.txt"
PY_SCRIPT="/sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/46-addition_2/node_4/generated_code.py"

# 1. Create conda environment if not exists
if [ ! -d "$ENV_DIR" ]; then
    conda create -y -p "$ENV_DIR" python=$PYTHON_VERSION
fi

# 2. Activate environment
eval "$(conda shell.bash hook)"
conda activate "$ENV_DIR"

# 3. Install uv if not present
if ! python -m uv --version >/dev/null 2>&1; then
    pip install uv
fi

# 4. Install requirements with uv (prerelease allowed, exact versions)
python -m uv pip install -r "$REQ_ML" --prerelease=allow -r "$REQ_COMMON"

# 5. Install additional packages needed for the script
python -m uv pip install pandas numpy scikit-learn lightgbm pillow tqdm torch torchvision

# 6. Run the Python script
python "$PY_SCRIPT"
```

**Usage:**  
Save as `run.sh`, make executable (`chmod +x run.sh`), and run:  
```bash
./run.sh
```

**Notes:**  
- Installs only the packages (and dependencies) specified in the requirements files, plus any additional packages needed for the script.
- Minimal and robust for your described workflow.
