import os

file_path = "/sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/addition_2_data/descriptions.txt"

def print_text_head(fp, max_chars=768):
    with open(fp, 'r', encoding='utf-8', errors='replace') as f:
        content = ''
        for line in f:
            if len(content) + len(line) > max_chars:
                content += line[:max_chars-len(content)]
                break
            content += line
        print(content)

def try_tabular(fp):
    import pandas as pd
    try:
        df = pd.read_csv(fp, nrows=3, dtype=str)
    except Exception:
        return False
    cols = list(df.columns)
    ncol = len(cols)
    if ncol > 20:
        shown = cols[:10] + ['...'] + cols[-10:]
    else:
        shown = cols
    print("Columns:", shown)
    pd.set_option('display.max_colwidth', 50)
    print(df.head(3).to_string(index=False))
    return True

try:
    if try_tabular(file_path):
        pass
    else:
        print_text_head(file_path)
except Exception:
    print_text_head(file_path)
