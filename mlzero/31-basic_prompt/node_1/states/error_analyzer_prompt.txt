def infer_label_column(df):
    """Infer the label column for binary classification."""

def map_label_to_binary(label):
    """Map string labels to binary: malignant=1, others=0."""

def extract_image_features(image_paths, device, batch_size=32):
    """
    Extract features from images using pretrained ResNet50 (avgpool output).
    Returns a numpy array of shape (len(image_paths), 2048).
    """
    model = models.resnet50(pretrained=True)
    model = torch.nn.Sequential(*(list(model.children())[:-1]))  # Remove FC layer
    model.eval()
    model.to(device)

    transform = T.Compose([
        T.Resize((224, 224)),
        T.ToTensor(),
        T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ])

    feats = []
    with torch.no_grad():
        for i in tqdm(range(0, len(image_paths), batch_size), desc="Extracting image features"):
            batch_paths = image_paths[i:i+batch_size]
            imgs = []
            for p in batch_paths:
                img = Image.open(p).convert('RGB')
                imgs.append(transform(img))
            imgs = torch.stack(imgs).to(device)
            out = model(imgs).squeeze(-1).squeeze(-1)  # (B, 2048, 1, 1) -> (B, 2048)
            feats.append(out.cpu().numpy())
    feats = np.concatenate(feats, axis=0)
    return feats

def get_tabular_features(df, fit_encoders=False, encoders=None, scaler=None):
    """
    Extract tabular features from dataframe.
    Returns numpy array, updated encoders and scaler.
    """
    tabular_cols = []
    # Use all columns except label, image_name, expert_opinion (which is mostly NaN), and index columns
    exclude = {'label', 'image_name', 'expert_opinion'}
    exclude.update([c for c in df.columns if 'unnamed' in c.lower() or 'index' in c.lower()])
    for c in df.columns:
        if c not in exclude:
            tabular_cols.append(c)
    X_tab = df[tabular_cols].copy()
    # Fill missing values
    X_tab = X_tab.fillna(-1)
    # Encode categorical columns
    if fit_encoders or encoders is None:
        encoders = {}
        for col in X_tab.select_dtypes(include=['object', 'category']).columns:
            le = LabelEncoder()
            X_tab[col] = le.fit_transform(X_tab[col].astype(str))
            encoders[col] = le
    else:
        for col in X_tab.select_dtypes(include=['object', 'category']).columns:
            le = encoders[col]
            X_tab[col] = le.transform(X_tab[col].astype(str))
    # Scale features
    if fit_encoders or scaler is None:
        scaler = StandardScaler()
        X_tab = scaler.fit_transform(X_tab)
    else:
        X_tab = scaler.transform(X_tab)
    return X_tab, encoders, scaler, tabular_cols
