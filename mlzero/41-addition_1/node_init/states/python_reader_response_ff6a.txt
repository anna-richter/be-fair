```python
import os
import pandas as pd

file_path = "/sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/addition_1_data/train.csv"

def display_tabular(df):
    cols = list(df.columns)
    n = len(cols)
    if n > 20:
        show_cols = cols[:10] + cols[-10:]
    else:
        show_cols = cols
    print("Columns:", show_cols)
    pd.set_option('display.max_colwidth', 50)
    print(df[show_cols].head(3).to_string(index=False))

def display_text(fp):
    with open(fp, 'r', encoding='utf-8', errors='ignore') as f:
        txt = f.read(768)
    print(txt)

try:
    df = pd.read_csv(file_path)
    display_tabular(df)
except Exception:
    display_text(file_path)
```
