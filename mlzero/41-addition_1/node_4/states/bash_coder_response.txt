```bash
#!/bin/bash
set -euo pipefail

# Paths
ENV_DIR="/sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/41-addition_1/node_4/conda_env"
REQ_COMMON="/home/anri21/.conda/envs/mlzero_env/lib/python3.11/site-packages/autogluon/assistant/tools_registry/_common/requirements.txt"
REQ_MM="/home/anri21/.conda/envs/mlzero_env/lib/python3.11/site-packages/autogluon/assistant/tools_registry/autogluon.multimodal/requirements.txt"
PY_SCRIPT="/sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/41-addition_1/node_4/generated_code.py"

# 1. Create conda env if not exists
if [ ! -d "$ENV_DIR" ]; then
    conda create -y -p "$ENV_DIR" python=3.11
fi

# 2. Activate conda env
# shellcheck disable=SC1091
source "$(conda info --base)/etc/profile.d/conda.sh"
conda activate "$ENV_DIR"

# 3. Install uv
pip install --upgrade pip
pip install uv

# 4. Install requirements with uv (prerelease allowed, exact requirements)
uv pip install --prerelease=allow -r "$REQ_MM" -r "$REQ_COMMON"

# 5. Run the Python script
python "$PY_SCRIPT"
```

**Instructions:**
1. Save as `run_node4.sh`
2. `chmod +x run_node4.sh`
3. Run: `./run_node4.sh`

**Notes:**
- This script only installs the exact packages (and dependencies) specified in the requirements files.
- It will not upgrade or reinstall `autogluon.multimodal` if the correct version is already present.
- If you want to always recreate the environment, add `conda remove -y -p "$ENV_DIR" --all` before creation.
