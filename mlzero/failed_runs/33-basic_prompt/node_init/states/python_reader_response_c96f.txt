```python
import os
import pandas as pd

file_path = "/home/anri21/be-fair/mlzero/basic_prompt_data/mydataset.csv"

def print_columns(cols):
    n = len(cols)
    if n > 20:
        print("Columns (first 10 + last 10):")
        print(list(cols[:10]) + ["..."] + list(cols[-10:]))
    else:
        print("Columns:", list(cols))

def print_rows(df):
    pd.set_option('display.max_colwidth', 50)
    rows = df.head(3)
    print(rows.to_string(index=False))

def try_tabular(path):
    try:
        df = pd.read_csv(path)
        print_columns(df.columns)
        print_rows(df)
        return True
    except Exception:
        return False

def try_text(path):
    try:
        with open(path, 'r', encoding='utf-8', errors='replace') as f:
            content = f.read(768)
            print("Text preview:\n", content)
        return True
    except Exception:
        return False

if not try_tabular(file_path):
    if not try_text(file_path):
        size = os.path.getsize(file_path)
        print(f"File type: binary or unknown. Size: {size} bytes")
```
