```bash
#!/bin/bash
set -e

ENV_DIR="/home/anri21/be-fair/mlzero/28-basic_prompt/node_14/conda_env"
PYTHON_VERSION="3.11"
REQ_COMMON="/home/anri21/.conda/envs/mlzero_env/lib/python3.12/site-packages/autogluon/assistant/tools_registry/_common/requirements.txt"
REQ_TABULAR="/home/anri21/.conda/envs/mlzero_env/lib/python3.12/site-packages/autogluon/assistant/tools_registry/autogluon.tabular/requirements.txt"
PY_SCRIPT="/home/anri21/be-fair/mlzero/28-basic_prompt/node_14/generated_code.py"

# 1. Create conda env if not exists
if [ ! -d "$ENV_DIR" ]; then
    conda create -y -p "$ENV_DIR" python=$PYTHON_VERSION
fi

# 2. Activate env
source "$(conda info --base)/etc/profile.d/conda.sh"
conda activate "$ENV_DIR"

# 3. Install uv
pip install --upgrade pip
pip install uv

# 4. Install requirements with uv (prerelease allowed, exact versions)
uv pip install --prerelease=allow -r "$REQ_TABULAR" -r "$REQ_COMMON"

# 5. Run the Python script
python "$PY_SCRIPT"
```

**Instructions:**
- Save as `run_node14.sh`, `chmod +x run_node14.sh`, and run with `./run_node14.sh`.
- This script only installs the exact packages (and dependencies) specified in the requirements files, and will not upgrade/reinstall `autogluon.tabular` if already at the correct version.
- Make sure your test CSV exists as required by your Python script.
