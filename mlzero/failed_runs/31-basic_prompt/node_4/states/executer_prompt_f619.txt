"""
Skin Lesion Malignancy Probability Prediction Script

This script trains a machine learning model (using image features and tabular data)
to predict the probability that a skin lesion is malignant, based on a dataset of lesion images and metadata.

1. Data preprocessing:
   - Loads the dataset and removes samples from the training set with missing labels.
   - Removes unnecessary index columns.
   - Removes training rows with missing image files (to avoid errors).
   - Extracts features from images (using pretrained ResNet50) and combines with tabular features.
   - Splits the data into training and validation sets.
2. Model training:
   - Trains a LightGBM classifier for binary classification.
   - Saves the trained model and feature pipeline to a timestamped folder in the specified output directory.
3. Prediction:
   - Makes malignancy probability predictions for the entire test set, preserving original indices.
   - Saves the results in the same format and extension as the test data, with correct column names.
4. Validation:
   - Computes and prints the AUROC score on the held-out validation set.
5. Utility:
   - Provides a function to predict malignancy probabilities for a folder of new images.

import os
import random
import time
import pandas as pd
import numpy as np

from tqdm import tqdm
from PIL import Image

import torch
import torchvision.transforms as T
import torchvision.models as models

from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score
from sklearn.preprocessing import LabelEncoder, StandardScaler
import lightgbm as lgb
import joblib

SEED = 42

# Paths
DATA_DIR = "/home/anri21/be-fair/mlzero/basic_prompt_data"
OUTPUT_DIR = "/home/anri21/be-fair/mlzero/31-basic_prompt/node_4/output"
TRAIN_CSV = os.path.join(DATA_DIR, "mydataset.csv")

def get_test_file_and_format():
    """
    Find the test file in the data directory and determine its extension.
    Returns (test_file_path, extension)
    """
    for fname in os.listdir(DATA_DIR):
        if fname.lower().startswith("test"):
            ext = os.path.splitext(fname)[1]
            return os.path.join(DATA_DIR, fname), ext
    raise FileNotFoundError("Test file not found in data directory.")

def get_sample_submission_file_and_format():
    """
    Find the sample submission file in the data directory and determine its extension.
    Returns (sample_submission_path, extension)
    """
    for fname in os.listdir(DATA_DIR):
        if "sample" in fname.lower() and "submission" in fname.lower():
            ext = os.path.splitext(fname)[1]
            return os.path.join(DATA_DIR, fname), ext
    return None, None

def make_timestamped_model_dir(base_dir):
    """Create a timestamped directory for saving the model."""
    ts = int(time.time() * 1000) + random.randint(0, 999)
    model_dir = os.path.join(base_dir, f"model_{ts}")
    os.makedirs(model_dir, exist_ok=True)
    return model_dir

def infer_label_column(df):
    """Infer the label column for binary classification."""
    if 'label' in df.columns:
        unique = df['label'].dropna().unique()
        if set(unique) <= set(['malignant', 'benign', 'non-neoplastic']):
            return 'label'
    for col in df.columns:
        if df[col].nunique() == 2:
            return col
    raise ValueError("Could not infer label column.")

def map_label_to_binary(label):
    """Map string labels to binary: malignant=1, others=0."""
    return 1 if str(label).strip().lower() == "malignant" else 0

def extract_image_features(image_paths, device, batch_size=32):
    """
    Extract features from images using pretrained ResNet50 (avgpool output).
    Returns a numpy array of shape (len(image_paths), 2048).
    """
    model = models.resnet50(pretrained=True)
    model = torch.nn.Sequential(*(list(model.children())[:-1]))  # Remove FC layer
    model.eval()
    model.to(device)

    transform = T.Compose([
        T.Resize((224, 224)),
        T.ToTensor(),
        T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ])

    feats = []
    with torch.no_grad():
        for i in tqdm(range(0, len(image_paths), batch_size), desc="Extracting image features"):
            batch_paths = image_paths[i:i+batch_size]
            imgs = []
            for p in batch_paths:
                if os.path.exists(p):
                    img = Image.open(p).convert('RGB')
                    imgs.append(transform(img))
                else:
                    # For missing images, fill with zeros
                    imgs.append(torch.zeros(3, 224, 224))
            imgs = torch.stack(imgs).to(device)
            out = model(imgs).squeeze(-1).squeeze(-1)  # (B, 2048, 1, 1) -> (B, 2048)
            feats.append(out.cpu().numpy())
    feats = np.concatenate(feats, axis=0)
    return feats

def get_tabular_features(df, fit_encoders=False, encoders=None, scaler=None):
    """
    Extract tabular features from dataframe.
    Returns numpy array, updated encoders and scaler.
    """
    tabular_cols = []
    # Use all columns except label, image_name, expert_opinion (which is mostly NaN), and index columns
    exclude = {'label', 'image_name', 'expert_opinion'}
    exclude.update([c for c in df.columns if 'unnamed' in c.lower() or 'index' in c.lower()])
    for c in df.columns:
        if c not in exclude:
            tabular_cols.append(c)
    X_tab = df[tabular_cols].copy()
    # Fill missing values
    X_tab = X_tab.fillna(-1)
    # Encode categorical columns
    if fit_encoders or encoders is None:
        encoders = {}
        for col in X_tab.select_dtypes(include=['object', 'category']).columns:
            le = LabelEncoder()
            X_tab[col] = le.fit_transform(X_tab[col].astype(str))
            encoders[col] = le
    else:
        for col in X_tab.select_dtypes(include=['object', 'category']).columns:
            le = encoders[col]
            X_tab[col] = le.transform(X_tab[col].astype(str))
    # Scale features
    if fit_encoders or scaler is None:
        scaler = StandardScaler()
        X_tab = scaler.fit_transform(X_tab)
    else:
        X_tab = scaler.transform(X_tab)
    return X_tab, encoders, scaler, tabular_cols

def main():
    # 1. Data Loading and Preprocessing
    # ---------------------------------
    df = pd.read_csv(TRAIN_CSV)
    # Remove unnecessary index column if present
    for idx_col in ['Unnamed: 0', 'index']:
        if idx_col in df.columns:
            df = df.drop(columns=[idx_col])
    # Remove training samples with missing labels
    label_col = infer_label_column(df)
    df = df.dropna(subset=[label_col]).reset_index(drop=True)
    # Map label to binary (malignant=1, others=0)
    df['label_binary'] = df[label_col].apply(map_label_to_binary)
    # Add absolute image path column
    df['image_path'] = df['image_name'].apply(lambda x: os.path.join(DATA_DIR, x))
    # Remove rows with missing image files (to avoid errors)
    missing_images = df[~df['image_path'].apply(os.path.exists)]
    if not missing_images.empty:
        print(f"Warning: Removing {len(missing_images)} training rows with missing image files.")
        df = df[df['image_path'].apply(os.path.exists)].reset_index(drop=True)
    # Check that there is at least one training sample left
    if len(df) == 0:
        raise ValueError("No training samples with valid images found after filtering.")

    # 2. Train/Validation Split
    # ------------------------
    train_df, val_df = train_test_split(
        df, test_size=0.1, random_state=SEED, stratify=df['label_binary']
    )
    train_df = train_df.reset_index(drop=True)
    val_df = val_df.reset_index(drop=True)

    # 3. Feature Extraction
    # ---------------------
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    # Image features
    train_img_feats = extract_image_features(train_df['image_path'].tolist(), device)
    val_img_feats = extract_image_features(val_df['image_path'].tolist(), device)
    # Tabular features
    X_tab_train, encoders, scaler, tabular_cols = get_tabular_features(train_df, fit_encoders=True)
    X_tab_val, _, _, _ = get_tabular_features(val_df, fit_encoders=False, encoders=encoders, scaler=scaler)
    # Combine features
    X_train = np.concatenate([train_img_feats, X_tab_train], axis=1)
    X_val = np.concatenate([val_img_feats, X_tab_val], axis=1)
    y_train = train_df['label_binary'].values
    y_val = val_df['label_binary'].values

    # 4. Model Training
    # -----------------
    model_dir = make_timestamped_model_dir(OUTPUT_DIR)
    clf = lgb.LGBMClassifier(
        n_estimators=200,
        random_state=SEED,
        class_weight='balanced'
    )
    clf.fit(X_train, y_train)
    # Save model and encoders
    joblib.dump(clf, os.path.join(model_dir, "model.pkl"))
    joblib.dump(encoders, os.path.join(model_dir, "encoders.pkl"))
    joblib.dump(scaler, os.path.join(model_dir, "scaler.pkl"))
    joblib.dump(tabular_cols, os.path.join(model_dir, "tabular_cols.pkl"))

    # 5. Prediction on Test Set
    # -------------------------
    test_csv, test_ext = get_test_file_and_format()
    test_df = pd.read_csv(test_csv)
    for idx_col in ['Unnamed: 0', 'index']:
        if idx_col in test_df.columns:
            test_df = test_df.drop(columns=[idx_col])
    test_df['image_path'] = test_df['image_name'].apply(lambda x: os.path.join(DATA_DIR, x))
    # For missing test images, fill with zeros (so that row count is preserved)
    missing_test_images = ~test_df['image_path'].apply(os.path.exists)
    if missing_test_images.any():
        print(f"Warning: {missing_test_images.sum()} test images missing. Filling features with zeros.")
    # Extract image features (for missing images, fill with zeros)
    test_img_feats = []
    batch_size = 32
    model = models.resnet50(pretrained=True)
    model = torch.nn.Sequential(*(list(model.children())[:-1]))
    model.eval()
    model.to(device)
    transform = T.Compose([
        T.Resize((224, 224)),
        T.ToTensor(),
        T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ])
    with torch.no_grad():
        for i in tqdm(range(0, len(test_df), batch_size), desc="Extracting test image features"):
            batch_paths = test_df['image_path'].iloc[i:i+batch_size].tolist()
            imgs = []
            for p in batch_paths:
                if os.path.exists(p):
                    img = Image.open(p).convert('RGB')
                    imgs.append(transform(img))
                else:
                    imgs.append(torch.zeros(3, 224, 224))  # Placeholder for missing image
            imgs = torch.stack(imgs).to(device)
            out = model(imgs).squeeze(-1).squeeze(-1)
            test_img_feats.append(out.cpu().numpy())
    test_img_feats = np.concatenate(test_img_feats, axis=0)
    # Tabular features
    X_tab_test, _, _, _ = get_tabular_features(test_df, fit_encoders=False, encoders=encoders, scaler=scaler)
    X_test = np.concatenate([test_img_feats, X_tab_test], axis=1)
    # Predict probabilities for the positive class (malignant)
    y_pred_proba = clf.predict_proba(X_test)[:, 1]

    # Prepare output DataFrame
    sample_sub_path, _ = get_sample_submission_file_and_format()
    if sample_sub_path:
        sample_sub = pd.read_csv(sample_sub_path)
        output_cols = sample_sub.columns.tolist()
    else:
        output_cols = [label_col]
    results_df = test_df.copy()
    results_df[output_cols[0]] = y_pred_proba
    results_df = results_df[[col for col in output_cols if col in results_df.columns] + [output_cols[0]]].copy()
    results_df = results_df.set_index(test_df.index)

    # Save results in the same format and extension as test file
    results_path = os.path.join(OUTPUT_DIR, f"results{test_ext}")
    if test_ext == ".csv":
        results_df.to_csv(results_path, index=False)
    elif test_ext in [".tsv", ".txt"]:
        results_df.to_csv(results_path, sep='\t', index=False)
    else:
        raise ValueError(f"Unsupported test file extension: {test_ext}")

    # 6. Validation
    # -------------
    try:
        val_pred_proba = clf.predict_proba(X_val)[:, 1]
        val_auc = roc_auc_score(y_val, val_pred_proba)
        print(f"Validation AUROC: {val_auc:.4f}")
    except Exception as e:
        print(f"Validation failed: {e}")

    # 7. Validation Checks
    # -------------------
    # a) Check output file exists
    assert os.path.exists(results_path), "Prediction file was not saved."
    # b) Check number of predictions matches test set
    pred_df = pd.read_csv(results_path)
    assert len(pred_df) == len(test_df), "Number of predictions does not match test set."
    # c) Check output columns match requirements
    assert output_cols[0] in pred_df.columns, f"Output column '{output_cols[0]}' missing in results."
    # d) Check indices preserved
    assert all(pred_df.index == test_df.index), "Test data indices not preserved in output."
    # e) Check predictions are floats in [0, 1]
    assert np.all((pred_df[output_cols[0]] >= 0) & (pred_df[output_cols[0]] <= 1)), "Predictions not in [0, 1]."

    print(f"Predictions saved to: {results_path}")

    # 8. Utility: Function to predict on new images
    # ---------------------------------------------
    def predict_folder(folder_path):
        """
        Given a folder path containing images, returns a DataFrame with image names and
        predicted malignancy probabilities (float in [0, 1]) for each image.
        """
        image_files = [
            f for f in os.listdir(folder_path)
            if os.path.isfile(os.path.join(folder_path, f)) and f.lower().endswith(('.jpg', '.jpeg', '.png'))
        ]
        if not image_files:
            raise ValueError("No image files found in the provided folder.")
        df_new = pd.DataFrame({'image_name': image_files})
        df_new['image_path'] = df_new['image_name'].apply(lambda x: os.path.join(folder_path, x))
        # Extract image features
        feats = []
        with torch.no_grad():
            for i in range(0, len(df_new), 32):
                batch_paths = df_new['image_path'].iloc[i:i+32].tolist()
                imgs = []
                for p in batch_paths:
                    if os.path.exists(p):
                        img = Image.open(p).convert('RGB')
                        imgs.append(transform(img))
                    else:
                        imgs.append(torch.zeros(3, 224, 224))
                imgs = torch.stack(imgs).to(device)
                out = model(imgs).squeeze(-1).squeeze(-1)
                feats.append(out.cpu().numpy())
        feats = np.concatenate(feats, axis=0)
        # No tabular features for new images, so fill with zeros
        X_tab = np.zeros((len(df_new), len(tabular_cols)))
        X = np.concatenate([feats, X_tab], axis=1)
        proba = clf.predict_proba(X)[:, 1]
        df_new['malignancy_probability'] = proba
        return df_new[['image_name', 'malignancy_probability']]

    # Example usage:
    # new_results = predict_folder("/path/to/new/images")
    # print(new_results.head())

if __name__ == "__main__":
    main()
