"""
Skin Lesion Malignancy Probability Prediction using AutoGluon Tabular

This script trains an AutoGluon Tabular model to predict the probability that a skin lesion is malignant,
using only tabular metadata (no image features). It performs the following steps:

1. Data preprocessing:
   - Loads the dataset and removes samples from the training set with missing labels.
   - Removes unnecessary index columns.
   - Drops columns that are not usable for tabular prediction (e.g., image_name, expert_opinion).
2. Model training:
   - Trains an AutoGluon Tabular predictor for binary classification using presets="extreme".
   - Saves the trained model to a timestamped folder in the specified output directory.
3. Prediction:
   - Makes malignancy probability predictions for the entire test set, preserving original indices.
   - Saves the results in the same format and extension as the test data, with correct column names.
4. Validation:
   - Prints the AUROC score on AutoGluon's internal validation set.
5. Utility:
   - Provides a function to predict malignancy probabilities for a folder of new images (returns NaN, as no image features are used).

Installation requirements (run before executing this script):
------------------------------------------------------------
# pip install autogluon.tabular pandas scikit-learn

Author: AutoML Agent
"""

# Installation steps (run in bash before running this script):
# pip install autogluon.tabular pandas scikit-learn

import os
import random
import time
import pandas as pd
import numpy as np

from autogluon.tabular import TabularPredictor

# Set random seed for reproducibility
SEED = 42

# Paths
DATA_DIR = "/home/anri21/be-fair/mlzero/basic_prompt_data"
OUTPUT_DIR = "/home/anri21/be-fair/mlzero/31-basic_prompt/node_2/output"
TRAIN_CSV = os.path.join(DATA_DIR, "mydataset.csv")

def get_test_file_and_format():
    """
    Find the test file in the data directory and determine its extension.
    Returns (test_file_path, extension)
    """
    for fname in os.listdir(DATA_DIR):
        if fname.lower().startswith("test"):
            ext = os.path.splitext(fname)[1]
            return os.path.join(DATA_DIR, fname), ext
    raise FileNotFoundError("Test file not found in data directory.")

def get_sample_submission_file_and_format():
    """
    Find the sample submission file in the data directory and determine its extension.
    Returns (sample_submission_path, extension)
    """
    for fname in os.listdir(DATA_DIR):
        if "sample" in fname.lower() and "submission" in fname.lower():
            ext = os.path.splitext(fname)[1]
            return os.path.join(DATA_DIR, fname), ext
    return None, None

def make_timestamped_model_dir(base_dir):
    """Create a timestamped directory for saving the model."""
    ts = int(time.time() * 1000) + random.randint(0, 999)
    model_dir = os.path.join(base_dir, f"model_{ts}")
    os.makedirs(model_dir, exist_ok=True)
    return model_dir

def infer_label_column(df):
    """Infer the label column for binary classification."""
    if 'label' in df.columns:
        unique = df['label'].dropna().unique()
        if set(unique) <= set(['malignant', 'benign', 'non-neoplastic']):
            return 'label'
    for col in df.columns:
        if df[col].nunique() == 2:
            return col
    raise ValueError("Could not infer label column.")

def map_label_to_binary(label):
    """Map string labels to binary: malignant=1, others=0."""
    return 1 if str(label).strip().lower() == "malignant" else 0

def main():
    # 1. Data Loading and Preprocessing
    # ---------------------------------
    df = pd.read_csv(TRAIN_CSV)
    # Remove unnecessary index column if present
    for idx_col in ['Unnamed: 0', 'index']:
        if idx_col in df.columns:
            df = df.drop(columns=[idx_col])
    # Remove training samples with missing labels
    label_col = infer_label_column(df)
    df = df.dropna(subset=[label_col]).reset_index(drop=True)
    # Map label to binary (malignant=1, others=0)
    df['label_binary'] = df[label_col].apply(map_label_to_binary)
    # Drop columns not usable for tabular prediction
    drop_cols = ['image_name', 'expert_opinion']
    drop_cols = [c for c in drop_cols if c in df.columns]
    df_tab = df.drop(columns=drop_cols)
    # Use label_binary as the label for training
    train_data = df_tab.copy()
    label = 'label_binary'

    # 2. Model Training
    # -----------------
    model_dir = make_timestamped_model_dir(OUTPUT_DIR)
    predictor = TabularPredictor(
        label=label,
        path=model_dir,
        problem_type='binary',
        eval_metric='roc_auc'
    )
    # Use all training data; AutoGluon handles internal validation
    predictor.fit(
        train_data=train_data,
        presets="extreme",
        time_limit=None,
        verbosity=2
    )
    # Save the model (already saved by predictor.fit, but ensure it's in the right place)
    predictor.save()

    # 3. Prediction on Test Set
    # -------------------------
    test_csv, test_ext = get_test_file_and_format()
    test_df = pd.read_csv(test_csv)
    for idx_col in ['Unnamed: 0', 'index']:
        if idx_col in test_df.columns:
            test_df = test_df.drop(columns=[idx_col])
    # Drop columns not usable for tabular prediction
    drop_cols = ['image_name', 'expert_opinion']
    drop_cols = [c for c in drop_cols if c in test_df.columns]
    test_tab = test_df.drop(columns=drop_cols)
    # Predict probabilities for the positive class (malignant)
    y_pred_proba = predictor.predict_proba(test_tab)[1].values  # Probability for class 1

    # Prepare output DataFrame
    sample_sub_path, _ = get_sample_submission_file_and_format()
    if sample_sub_path:
        sample_sub = pd.read_csv(sample_sub_path)
        output_cols = sample_sub.columns.tolist()
    else:
        output_cols = [label_col]
    results_df = test_df.copy()
    results_df[output_cols[0]] = y_pred_proba
    results_df = results_df[[col for col in output_cols if col in results_df.columns] + [output_cols[0]]].copy()
    results_df = results_df.set_index(test_df.index)

    # Save results in the same format and extension as test file
    results_path = os.path.join(OUTPUT_DIR, f"results{test_ext}")
    if test_ext == ".csv":
        results_df.to_csv(results_path, index=False)
    elif test_ext in [".tsv", ".txt"]:
        results_df.to_csv(results_path, sep='\t', index=False)
    else:
        raise ValueError(f"Unsupported test file extension: {test_ext}")

    # 4. Validation
    # -------------
    try:
        # AutoGluon prints validation scores after fit; print again for clarity
        leaderboard = predictor.leaderboard(silent=True)
        print("AutoGluon internal validation leaderboard:")
        print(leaderboard)
        best_score = leaderboard.loc[leaderboard['score_val'].idxmax(), 'score_val']
        print(f"Best internal validation AUROC: {best_score:.4f}")
    except Exception as e:
        print(f"Validation failed: {e}")

    # 5. Validation Checks
    # -------------------
    # a) Check output file exists
    assert os.path.exists(results_path), "Prediction file was not saved."
    # b) Check number of predictions matches test set
    pred_df = pd.read_csv(results_path)
    assert len(pred_df) == len(test_df), "Number of predictions does not match test set."
    # c) Check output columns match requirements
    assert output_cols[0] in pred_df.columns, f"Output column '{output_cols[0]}' missing in results."
    # d) Check indices preserved
    assert all(pred_df.index == test_df.index), "Test data indices not preserved in output."
    # e) Check predictions are floats in [0, 1]
    assert np.all((pred_df[output_cols[0]] >= 0) & (pred_df[output_cols[0]] <= 1)), "Predictions not in [0, 1]."

    print(f"Predictions saved to: {results_path}")

    # 6. Utility: Function to predict on new images (returns NaN, as no image features are used)
    # ------------------------------------------------------------------------------------------
    def predict_folder(folder_path):
        """
        Given a folder path containing images, returns a DataFrame with image names and
        predicted malignancy probabilities (float in [0, 1]) for each image.
        Since this model does not use image features, returns NaN for all images.
        """
        image_files = [
            f for f in os.listdir(folder_path)
            if os.path.isfile(os.path.join(folder_path, f)) and f.lower().endswith(('.jpg', '.jpeg', '.png'))
        ]
        if not image_files:
            raise ValueError("No image files found in the provided folder.")
        df_new = pd.DataFrame({'image_name': image_files})
        df_new['malignancy_probability'] = np.nan
        return df_new[['image_name', 'malignancy_probability']]

    # Example usage:
    # new_results = predict_folder("/path/to/new/images")
    # print(new_results.head())

if __name__ == "__main__":
    main()
