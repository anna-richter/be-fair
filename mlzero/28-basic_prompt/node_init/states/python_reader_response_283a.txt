```python
import os
import pandas as pd

file_path = "/sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/basic_prompt_data/test.csv"

def print_columns(cols):
    n = len(cols)
    if n > 20:
        print("Columns:", list(cols[:10]) + ["..."] + list(cols[-10:]))
    else:
        print("Columns:", list(cols))

def truncate_cell(val):
    s = str(val)
    return s if len(s) <= 50 else s[:47] + "..."

def show_tabular(df):
    print_columns(df.columns)
    rows = df.head(3)
    print(rows.applymap(truncate_cell).to_string(index=False))

def show_text(fp):
    with open(fp, 'r', encoding='utf-8', errors='replace') as f:
        txt = f.read(768)
        print(txt)

try:
    df = pd.read_csv(file_path)
    show_tabular(df)
except Exception:
    try:
        show_text(file_path)
    except Exception as e:
        print(f"Could not read file: {e}")
```
