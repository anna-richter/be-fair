"""
Skin Lesion Malignancy Probability Prediction using AutoGluon Tabular

This script trains an AutoGluon TabularPredictor to predict the probability that a skin lesion is malignant,

It:
- Loads and preprocesses the data (removes NA labels, drops index columns, maps 'benign' to 'non-neoplastic').
- Trains a TabularPredictor with presets="extreme" and eval_metric="roc_auc".
- Predicts malignancy probability for the test set, preserving original indices and output format.
- Saves the trained model and prediction results to the specified output directory.
- Performs validation checks to ensure output correctness.

Installation requirements (run in bash before running this script):
    pip install autogluon.tabular[all] pandas numpy

Usage:
    Place this script in any location and run it. It expects the following files:
        - /sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/basic_prompt_data/train.csv
        - /sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/basic_prompt_data/test.csv

Outputs:
    - Model directory: /sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/28-basic_prompt/node_2/output/<random_timestamp>_autogluon_tabular/
    - Prediction file: /sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/28-basic_prompt/node_2/output/results.csv (format matches test.csv)

# Installation steps (run in bash before running this script)
# pip install autogluon.tabular[all] pandas numpy

import os
import random
import time
import pandas as pd
import numpy as np

from autogluon.tabular import TabularPredictor

if __name__ == "__main__":
    # Paths
    DATA_DIR = "/sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/basic_prompt_data"
    TRAIN_CSV = os.path.join(DATA_DIR, "train.csv")
    TEST_CSV = os.path.join(DATA_DIR, "test.csv")
    OUTPUT_DIR = "/sc-scratch/sc-scratch-ikim-guidlight/be-fair/mlzero/28-basic_prompt/node_2/output"
    os.makedirs(OUTPUT_DIR, exist_ok=True)

    # 1. Data Loading and Preprocessing
    train = pd.read_csv(TRAIN_CSV)
    test = pd.read_csv(TEST_CSV)

    # Remove unnecessary index column if present
    for col in ['Unnamed: 0', 'index']:
        if col in train.columns:
            train = train.drop(columns=[col])
        if col in test.columns:
            test = test.drop(columns=[col])

    # Remove training samples without valid labels (drop NA in label column)
    train = train.dropna(subset=['label'])

    # Map 'benign' to 'non-neoplastic' if present
    train['label'] = train['label'].replace({'benign': 'non-neoplastic'})

    # Only keep 'malignant' and 'non-neoplastic'
    train = train[train['label'].isin(['malignant', 'non-neoplastic'])].reset_index(drop=True)

    # 2. Model Training
    # Save model to a random timestamped folder in OUTPUT_DIR
    timestamp = int(time.time()) + random.randint(0, 99999)
    model_dir = os.path.join(OUTPUT_DIR, f"{timestamp}_autogluon_tabular")
    os.makedirs(model_dir, exist_ok=True)

    # Use all columns except 'image_name' as features (but keep 'image_name' if you want to use it as a categorical feature)
    # We'll use all columns except 'label' as features
    feature_cols = [col for col in train.columns if col != 'label']

    # Train TabularPredictor
    predictor = TabularPredictor(
        label='label',
        path=model_dir,
        eval_metric='roc_auc'
    ).fit(
        train_data=train,
        presets="extreme",
        verbosity=2
    )

    # 3. Prediction on Test Set
    # AutoGluon will handle missing values in test automatically
    # Predict_proba returns a DataFrame with columns for each class
    proba_df = predictor.predict_proba(test)

    # The probability of 'malignant' is needed (float between 0 and 1)
    if 'malignant' in proba_df.columns:
        malignancy_proba = proba_df['malignant'].values
    else:
        raise ValueError(f"'malignant' class not found in prediction columns: {proba_df.columns}")

    # Prepare output DataFrame: must match test.csv format and row order
    results_df = test.copy()
    results_df['malignant'] = malignancy_proba

    # Save results with same format/extension as test.csv
    results_path = os.path.join(OUTPUT_DIR, "results.csv")
    results_df_out = results_df[['malignant']]
    results_df_out.to_csv(results_path, index=False)

    # 4. Validation checks
    # a) Check number of predictions matches test set
    assert len(results_df_out) == len(test), f"Number of predictions ({len(results_df_out)}) does not match test set ({len(test)})"
    # b) Check output columns
    assert list(results_df_out.columns) == ['malignant'], f"Output columns {list(results_df_out.columns)} do not match required ['malignant']"
    # c) Check output format (CSV)
    assert results_path.endswith('.csv'), "Output file must be CSV"
    # d) Check indices preserved (row order matches test set)
    assert all(results_df_out.index == test.index), "Output indices do not match test set indices"
    # e) Check probabilities are floats in [0,1]
    assert np.all((results_df_out['malignant'] >= 0) & (results_df_out['malignant'] <= 1)), "Malignancy probabilities are not all in [0,1]"

    # 5. Validation metric on internal validation set (AutoGluon reports this automatically)
    # Print leaderboard for validation performance
    try:
        leaderboard = predictor.leaderboard(silent=True)
        print("AutoGluon internal validation leaderboard:")
        print(leaderboard)
    except Exception as e:
        print(f"Validation failed: {e}")

    print(f"Model saved to: {model_dir}")
    print(f"Predictions saved to: {results_path}")
